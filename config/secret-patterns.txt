# Patterns for things that must never be committed. One extended regular expression per line.
# Used by scripts/check-secrets.sh and by the worker before it pushes (it scans only new lines).
# Text after two spaces and a # is a comment.
gh[pousr]_[A-Za-z0-9]{20,}  # GitHub tokens
github_pat_[A-Za-z0-9_]{20,}  # GitHub fine-grained tokens
sk-[A-Za-z0-9_-]{20,}  # OpenAI and Anthropic style keys
xox[abprs]-[A-Za-z0-9-]{10,}  # Slack
AKIA[0-9A-Z]{16}  # AWS access key id
AIza[0-9A-Za-z_-]{30,}  # Google API key
-----BEGIN [A-Z ]*PRIVATE KEY-----  # private keys
tskey-[A-Za-z0-9-]{10,}  # Tailscale auth key
swb_[A-Za-z0-9_-]{20,}  # swarm bot tokens
swt_[A-Za-z0-9_-]{20,}  # swarm task tokens
[a-z0-9-]+\.tail[a-z0-9]+\.ts\.net  # real tailnet hostnames
